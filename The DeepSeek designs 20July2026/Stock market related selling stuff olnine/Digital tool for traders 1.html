📈 TradingView Indicators — Pine Script Code Preview

⚡ SuperTrend Indicator • Pine Script v5 
//@version=5
indicator("SuperTrend Pro", overlay=true)

// Input Parameters
atrPeriod = input.int(10, "ATR Period")
factor = input.float(3.0, "Factor", step=0.1)

// SuperTrend Calculation
[supertrend, direction] = ta.supertrend(factor, atrPeriod)

// Plot
bullish = direction < 0
plot(bullish ? supertrend : na, "Up Trend", color=color.green, linewidth=2)
plot(not bullish ? supertrend : na, "Down Trend", color=color.red, linewidth=2)

// Buy/Sell Signals
plotshape(ta.crossover(direction, 0), "BUY", color=color.green, style=shape.labelup)
plotshape(ta.crossunder(direction, 0), "SELL", color=color.red, style=shape.labeldown)
